    function limparMascaraIdGenerico(element) {
        $(element).off('input.idgener');
        $(element).removeAttr('maxlength');

    function isValidIdGenericoPartial(value) {
        if (value === '') {
            return true;
        }
        if (/^[A-Z0-9]{1,9}$/.test(value)) {
            return true;
        }
        if (/^UX[A-Z0-9]{0,9}$/.test(value)) {
            return true;
        }
        var estrangPrefix = 'ESTRANG';
        if (estrangPrefix.indexOf(value) === 0) {
            return true;
        }
        return /^ESTRANG[A-Z0-9]{0,29}$/.test(value);
    }

    function formatarIdGenerico(value) {
        value = (value || '').toUpperCase().replace(/[^A-Z0-9]/g, '');
        var result = '';
        for (var i = 0; i < value.length; i++) {
            var candidate = result + value.charAt(i);
            if (isValidIdGenericoPartial(candidate)) {
                result = candidate;
            }
        }
        return result;
    }
